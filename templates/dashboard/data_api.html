Quickstart Integration

curl -X GET "https://agrilens.rw/api/v1/crops/?season=2024-2025" \
     -H "Accept: application/json"
import requests

url = "https://agrilens.rw/api/v1/crops/?season=2024-2025"
headers = {"Accept": "application/json"}
response = requests.get(url, headers=headers)
data = response.json()
print(f"Loaded {data['count']} priority crops.")
const response = await fetch('/api/v1/crops/?season=2024-2025', {
  headers: { 'Accept': 'application/json' }
});
const data = await response.json();
console.log('Agricultural crops count:', data.count);
